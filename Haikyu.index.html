import random import time # ========================================== # 1. 遊戲資料設定與角色圖像（統一使用雙位元 Emoji） # ========================================== # 為四位三年級生配置更具代表性的 Emoji 圖像 AVAIABLE_CHARACTERS = { "1": {"name": "澤村大地", "symbol": "🏔️ 大地", "icon": "🏔️"}, # 穩重如山 "2": {"name": "菅原孝之", "symbol": "😊 菅原", "icon": "😊"}, # 爽朗溫暖的微笑！ "3": {"name": "東峰旭", "symbol": "🧔 東峰", "icon": "🧔"}, # 溫柔的鬍子王牌 "4": {"name": "清水潔子", "symbol": "👓 潔子", "icon": "👓"} # 美麗的眼鏡經理 } # 隨機產生 16 格地圖（0號格固定為起點，隨機放置 3 個機會 Chance） money_effects = [200, 500, -300, 150, -100, 200, -600, 150, 500, -300, 200, 150, 500, -600] random.shuffle(money_effects) board = [0] * 16 board[0] = "START" chance_positions = random.sample(range(1, 16), 3) money_idx = 0 for i in range(1, 16): if i in chance_positions: board[i] = "Chance" else: board[i] = money_effects[money_idx] money_idx += 1 chance_cards = [ {"text": "發動快攻！前進 1 步", "value": 1}, {"text": "完美的接球！前進 2 步", "value": 2}, {"text": "跳發得分！前進 3 步", "value": 3}, {"text": "被對方攔網成功... 後退 2 步", "value": -2}, {"text": "發球失誤送分。後退 3 步", "value": -3}, {"text": "輪轉站位錯誤！大迷路後退 5 步", "value": -5} ] # ========================================== # 2. 地圖繪製函式 (4x4 正方形) # ========================================== def draw_map(board, players): """在終端機繪製 4x4 的大富翁地圖，並動態顯示角色圖像""" # 建立一個長度 16 的列表，存放每一格目前的玩家圖像 grid_occupants = [""] * 16 for p in players: grid_occupants[p['position']] += p['icon'] def fmt(idx): occupants = grid_occupants[idx] if occupants: # 如果有玩家，優先顯示頭像（保持長度對齊） return f"{occupants:<4}" else: val = board[idx] if val == "START": return "🏁 " if val == "Chance": return "❓ " if val > 0: return f"+{val:<3}" if val < 0: return f"-{abs(val):<3}" return "0 " print("\n" + "="*40) print(f" 🏐 體育館目前戰況地圖 🏐") print("-" * 40) print(f"[{fmt(0)}] [{fmt(1)}] [{fmt(2)}] [{fmt(3)}]") print(f"[{fmt(15)}] [{fmt(4)}]") print(f"[{fmt(14)}] [{fmt(5)}]") print(f"[{fmt(13)}] [{fmt(12)}] [{fmt(11)}] [{fmt(10)}]") print("="*40) # ========================================== # 3. 遊戲初始化介面 # ========================================== print("==========================================") print("🏐 歡迎來到《排球少年》烏野三年級生大富翁！ 🏐") print("==========================================") while True: num_input = input("請選擇遊玩人數 (2、3、4 人): ").strip() if num_input in ["2", "3", "4"]: num_players = int(num_input) break print("❌ 輸入錯誤，請輸入 2、3 或 4。") players = [] pool = AVAIABLE_CHARACTERS.copy() for i in range(num_players): print(f"\n--- 請【玩家 {i+1}】選擇你的角色 ---") for key, char in pool.items(): print(f"[{key}] {char['name']} (頭像: {char['icon']})") while True: choice = input("請輸入角色編號: ").strip() if choice in pool: selected = pool.pop(choice) players.append({ "id": i, "name": selected["name"], "symbol": selected["symbol"], "icon": selected["icon"], "position": 0, "money": 300 }) print(f"✅ 成功選擇：{selected['name']} {selected['icon']} ！") break else: print("❌ 編號不存在，或該角色已被選走。") # ========================================== # 4. 遊戲主迴圈 # ========================================== game_over = False winner = None round_count = 1 while not game_over: print(f"\n======== 🏐 【 第 {round_count} 回合 開始 】 🏐 ========") draw_map(board, players) for player in players: input(f"\n👉 按 Enter 讓 {player['symbol']} 擲骰子...") dice = random.randint(1, 6) print(f"🎲 {player['symbol']} 擲出了 {dice} 點！") old_pos = player['position'] new_pos = old_pos + dice if new_pos >= 16: player['money'] += 100 print(f"💰 成功繞場一圈！獲得前輩應援 100 元！") player['position'] = new_pos % 16 current_grid = player['position'] print(f"🏃 移動到第 {current_grid} 格。") event_triggered = True while event_triggered: grid_effect = board[current_grid] if grid_effect == "Chance": print("✨ 踩到【機會】格！正在抽卡...") time.sleep(0.5) card = random.choice(chance_cards) print(f"🃏 機會卡：【{card['text']}】") old_pos = player['position'] new_pos = old_pos + card['value'] if card['value'] > 0 and new_pos >= 16: player['money'] += 100 print(f"💰 衝過起點！獲得前輩應援 100 元！") player['position'] = new_pos % 16 current_grid = player['position'] print(f"🚀 位置變更！移動到了第 {current_grid} 格。") event_triggered = True elif current_grid == 0: print("休息區（起點），儲備體力中。") event_triggered = False else: player['money'] += grid_effect if grid_effect >= 0: print(f"💵 獲得 {grid_effect} 元。") else: print(f"💸 失去 {abs(grid_effect)} 元。") event_triggered = False if player['money'] < 0: player['money'] = 0 print(f"💰 {player['symbol']} 目前總資產：{player['money']} 元。") print("-" * 30) if player['money'] >= 1200: game_over = True winner = player['name'] winner_icon = player['icon'] break round_count += 1 # ========================================== # 5. 遊戲結束 # ========================================== print("\n==========================================") print(f"🎉 🎉 🎉 飛吧！遊戲結束！🎉 🎉 🎉") print(f"恭喜 【{winner} {winner_icon}】 率先達到 1200 元，贏得本場比賽！") print("==========================================") for p in players: print(f"🏆 {p['name']} ({p['icon']}): {p['money']} 元")
